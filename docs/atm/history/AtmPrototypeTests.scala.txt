package effekt

import effekt.atm.{Checker, Lowering, Parser, Syntax}
import java.nio.file.{Files, Path}

class AtmTests extends munit.FunSuite {
  // Source mutations match tokens across formatting, and must actually change
  // the fixture: a no-op replacement would silently lose a rejection witness.
  extension (source: String) {
    private def replaceSyntax(before: String, after: String): String = {
      val tokens = "=>|[a-zA-Z_][a-zA-Z0-9_]*|[0-9]+|[^\\s]".r.findAllIn(before).toList
      val pattern = ("(?<![a-zA-Z0-9_])" + tokens.map(java.util.regex.Pattern.quote).mkString("\\s*") +
        "(?![a-zA-Z0-9_])").r
      assert(pattern.findFirstIn(source).nonEmpty, s"Missing source fragment: $before")
      pattern.replaceAllIn(source, _ => java.util.regex.Matcher.quoteReplacement(after))
    }
  }
  private def fixtureBody(name: String): String = {
    val input = Files.readString(Path.of("examples", "atm", s"$name.atm"))
    val prefix = "def main(): Bool = {"
    val start = input.indexOf(prefix)
    assert(start >= 0)
    input.substring(start + prefix.length, input.lastIndexOf('}'))
  }

  private def checked(name: String): Checker.CheckedProgram = {
    val input = Files.readString(Path.of("examples", "atm", s"$name.atm"))
    val result = Parser.parse(input).flatMap(Checker.check)
    result.fold(error => fail(error.message), identity)
  }

  private def execute(program: Checker.CheckedProgram, optimized: Boolean): String =
    AtmBackend.run(program, optimized).fold(fail(_), identity)

  for (name, expected) <- List("int-to-bool" -> "false", "nested-outer" -> "false",
    "file-protocol" -> "Open", "multishot" -> "true", "higher-order" -> "42", "stored-resumption" -> "42",
    "effectful-callback" -> "false", "return-clause" -> "true", "return-effects" -> "false",
    "callback-under-handler" -> "false", "capability-argument" -> "false",
    "large-integer" -> "9007199254740995", "capability-polymorphic" -> "true",
    "polymorphic-callback" -> "false", "effect-polymorphic" -> "false",
    "repeatable-effect" -> "88", "composed-effects" -> "200",
    "multiple-capabilities" -> "false") do {
    test(s"$name: ATM checking, Effekt compilation and JavaScript execution") {
      val program = checked(name)
      assertEquals(AtmReference.run(program.source), expected)
      assertEquals(execute(program, optimized = false), expected)
      assertEquals(execute(program, optimized = true), expected)
    }
  }

  test("a resumption depending on an outer capability cannot escape that capability") {
    val input = """def main(): ((Int) => Int) = {

  test("contextual lifting preserves all caller-owned answer layers at depths one through four") {
    import Syntax.*
    import Syntax.Value.*
    import Syntax.Term.*
    val int = Computation(Type.Int)
    val bool = Computation(Type.Bool)
    val base = Computation(Type.Int, Effect.Change(int, bool))
    val ask = Operation("ask", Type.Unit, Type.Int, int, bool, "arg", "k",
      Let("n", Apply(Variable("k"), Integer(42)), Return(Primitive("zero", List(Variable("n"))))))
    val callback = Lambda("u", Type.Unit, base, Perform("h", "ask", Unit))
    for depth <- 1 to 4 do {
      val pairs = (1 to depth).scanLeft((int, bool)) { case ((a, b), _) =>
        (Computation(Type.Int, Effect.Change(a, a)), Computation(Type.Int, Effect.Change(a, b)))
      }
      val call: Term = Apply(Variable("callback"), Unit)
      val nested = (1 to depth).reverse.foldLeft(call) { (body, index) =>
        val answer = pairs(index)._2
        val tick = Operation("tick", Type.Unit, Type.Unit, answer, answer, "arg", "k",
          Apply(Variable("k"), Unit))
        Handle(s"inner$index", answer, List(tick), body, "x", Return(Variable("x")))
      }
      val source = Program(bool, Handle("h", bool, List(ask),
        Let("callback", Return(callback), nested), "x", Return(Variable("x"))))
      val program = Checker.check(source).fold(error => fail(s"Depth $depth: ${error.message}"), identity)
      assertEquals(AtmReference.run(source), "false", s"source semantics at depth $depth")
      assertEquals(execute(program, optimized = false), "false", s"depth $depth")
      assertEquals(execute(program, optimized = true), "false", s"depth $depth")
    }
  }

  private val askContract = "{ def ask(): Int / [Int => Bool]; }"
  private val invoke = s"""capfun (c: $askContract) { fun (u: Unit): Int / [Int => Bool] { c.ask() } }"""

  private def checkText(input: String): Checker.CheckedProgram =
    Parser.parse(input).flatMap(Checker.check).fold(error => fail(error.message), identity)

  private def runText(input: String, expected: String): Unit = {
    val program = checkText(input)
    assertEquals(AtmReference.run(program.source), expected)
    assertEquals(execute(program, optimized = false), expected)
    assertEquals(execute(program, optimized = true), expected)
  }

  private def rejectText(input: String, message: String): Unit = {
    val parsed = Parser.parse(input).fold(error => fail(error.message), identity)
    val result = Checker.check(parsed)
    assert(result.isLeft, "Ill-typed source was accepted")
    assert(result.swap.toOption.get.message.contains(message), result.toString)
  }

  test("a capability abstraction instantiates with an abstract outer delimiter tail") {
    val body = fixtureBody("int-to-bool").replaceSyntax("h.ask()", "((invoke){cap(h)})()")

            x

  test("composition implications agree with an independent finite answer-state model") {
    // This model has two answer states and a polymorphic pure identity. It
    // intentionally does not call the checker's algebra or equality helpers.
    type Arrow = Option[(Int, Int)]
    val domain: List[Arrow] = None :: (for a <- List(0, 1); b <- List(0, 1) yield Some((a, b)))
    def followedBy(a: Arrow, b: Arrow): Either[Unit, Arrow] = (a, b) match {
      case (None, x) => Right(x)
      case (x, None) => Right(x)
      case (Some((middle, out)), Some((in, middle2))) if middle == middle2 => Right(Some((in, out)))
      case _ => Left(())
    }
    enum Expr {
      case Variable(name: String)
      case Then(a: Expr, b: Expr)
    }
    def render(e: Expr): String = e match {
      case Expr.Variable(name) => name
      case Expr.Then(a, b) => s"(${render(a)}; ${render(b)})"
    }
    def evaluate(e: Expr, assignment: Map[String, Arrow]): Either[Unit, Arrow] = e match {
      case Expr.Variable(name) => Right(assignment(name))
      case Expr.Then(a, b) => for x <- evaluate(a, assignment); y <- evaluate(b, assignment); z <- followedBy(x, y) yield z
    }
    val atoms = List("E", "F", "G").map(Expr.Variable.apply)
    val expressions = (for a <- atoms; b <- atoms yield Expr.Then(a, b)) ++
      (for a <- atoms; b <- atoms; c <- atoms; e <- List(Expr.Then(a, Expr.Then(b, c)), Expr.Then(Expr.Then(a, b), c)) yield e)
    val assignments = for {
      e <- domain; f <- domain; g <- domain
      prefix <- followedBy(e, f).toOption
      if followedBy(prefix, g).isRight
    } yield Map("E" -> e, "F" -> f, "G" -> g)
    var accepted = 0
    for expression <- expressions do {
      val input = s"""def main(): Int = {
        val helper = effectfun [E] {
          effectfun [F: after(E)] {
            effectfun [G: after((E;
            F))] {
              fun (callback: ((Unit) => Int / ${render(expression)})): Int { 0}
            }
          }
        };
        0
      }"""
      val parsed = Parser.parse(input)
      assert(parsed.isRight, parsed.toString)
      if Checker.check(parsed.toOption.get).isRight then {
        accepted += 1
        for assignment <- assignments do
          assert(evaluate(expression, assignment).isRight, s"Invalid implication: ${render(expression)} under $assignment")
      }
    }
    assert(accepted >= 4, s"Expected valid pairs and both associations of E;F;G, found $accepted")

    def syntax(arrow: Arrow): String = arrow match {
      case None => "pure"
      case Some((a, b)) => s"[State$a => State$b]"
    }
    for e <- domain; f <- domain do {
      val input = s"""def main(): Int = { val helper = (($compositionHelper)[effect ${syntax(e)}])[effect ${syntax(f)}]; 0 }"""
      val parsed = Parser.parse(input)
      assert(parsed.isRight, parsed.toString)
      assertEquals(Checker.check(parsed.toOption.get).isRight, followedBy(e, f).isRight,
        s"Concrete instantiation: $e then $f")
    }
  }

  private val identityContract = "{ def ask(): Int / [Int => Int]; }"
  private def identityHandler(name: String, body: String): String =
    s"""try { $body } with $name: Int { def ask(x: Unit): Int / [Int => Int] resume k = { k(42) } return x => { x } }"""
  private val scopedPair = s"""capfun (c: $identityContract) { capfun here (d: $identityContract) { fun (u: Unit): Int { 42 } } }"""
